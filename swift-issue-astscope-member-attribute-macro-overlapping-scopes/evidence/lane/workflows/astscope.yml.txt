name: astscope
on:
  workflow_dispatch:
permissions:
  contents: read
jobs:
  windows:
    runs-on: windows-latest
    steps:
      - uses: actions/checkout@3d3c42e5aac5ba805825da76410c181273ba90b1
      - uses: SwiftyLab/setup-swift@38f54a76b70d989321de9dc7c840618c08cf56e9
        with:
          swift-version: "6.4.1"
          cache-snapshot: false
      - shell: bash
        run: |
          astscope/run.sh windows-6.4.1
          astscope/min.sh windows-6.4.1
  macos:
    runs-on: macos-26
    steps:
      - uses: actions/checkout@3d3c42e5aac5ba805825da76410c181273ba90b1
      - name: Install swiftly and toolchains
        run: |
          curl -fsSL -o "$RUNNER_TEMP/swiftly.pkg" https://download.swift.org/swiftly/darwin/swiftly.pkg
          installer -pkg "$RUNNER_TEMP/swiftly.pkg" -target CurrentUserHomeDirectory
          "$HOME/.swiftly/bin/swiftly" init --assume-yes --skip-install --quiet-shell-followup
          for t in main-snapshot-2026-10-17 6.4.x-snapshot-2026-10-20 6.4.1; do "$HOME/.swiftly/bin/swiftly" install $t --assume-yes; done
      - name: Xcode
        run: |
          xcodebuild -version
          SWIFTC="xcrun swiftc" astscope/run.sh xcode
          SWIFTC="xcrun swiftc" astscope/min.sh xcode
      - name: swiftly toolchains
        run: |
          for t in main-snapshot-2026-10-17 6.4.x-snapshot-2026-10-20 6.4.1; do
            TOOLCHAIN=$t SWIFTC="$PWD/astscope/swiftly-swiftc.sh" astscope/run.sh macos-$t
            TOOLCHAIN=$t SWIFTC="$PWD/astscope/swiftly-swiftc.sh" astscope/min.sh macos-$t
          done
  linux:
    runs-on: ubuntu-latest
    strategy:
      fail-fast: false
      matrix:
        image: ["swiftlang/swift:nightly-main-jammy", "swift:6.4.1-noble"]
    container: ${{ matrix.image }}
    steps:
      - uses: actions/checkout@3d3c42e5aac5ba805825da76410c181273ba90b1
      - run: |
          astscope/run.sh "linux-${{ matrix.image }}"
          astscope/min.sh "linux-${{ matrix.image }}"
  original-macos-snapshot:
    if: github.event_name == 'workflow_dispatch'
    runs-on: macos-26
    timeout-minutes: 90
    steps:
      - name: Install swiftly and the main snapshot
        run: |
          curl -fsSL -o "$RUNNER_TEMP/swiftly.pkg" https://download.swift.org/swiftly/darwin/swiftly.pkg
          installer -pkg "$RUNNER_TEMP/swiftly.pkg" -target CurrentUserHomeDirectory
          "$HOME/.swiftly/bin/swiftly" init --assume-yes --skip-install --quiet-shell-followup
          "$HOME/.swiftly/bin/swiftly" install main-snapshot-2026-10-17 --assume-yes
      - name: Build swift-recursion a454a8a tests with the assertions snapshot
        run: |
          git clone -q https://github.com/swift-atoms/swift-recursion.git r
          cd r
          git checkout -q a454a8a6f9d5215e6f40ac64afa8a4bf29f2d643
          "$HOME/.swiftly/bin/swiftly" run swift --version +main-snapshot-2026-10-17
          set +e
          "$HOME/.swiftly/bin/swiftly" run swift build --build-tests +main-snapshot-2026-10-17 > build.log 2>&1
          echo "RESULT original-macos-main-snapshot exit=$?"
          grep -n -m3 -B3 -A14 'overlaps previous child' build.log
          grep -n -m10 'error:' build.log
          tail -5 build.log
  original-linux-nightly:
    if: github.event_name == 'workflow_dispatch'
    runs-on: ubuntu-latest
    timeout-minutes: 90
    container: swiftlang/swift:nightly-main-jammy
    steps:
      - name: Build swift-recursion a454a8a tests with the assertions nightly
        run: |
          git clone -q https://github.com/swift-atoms/swift-recursion.git r
          cd r
          git checkout -q a454a8a6f9d5215e6f40ac64afa8a4bf29f2d643
          swift --version
          set +e
          swift build --build-tests > build.log 2>&1
          echo "RESULT original-linux-nightly exit=$?"
          grep -n -m3 -B3 -A14 'overlaps previous child' build.log
          grep -n -m10 'error:' build.log
          tail -5 build.log
