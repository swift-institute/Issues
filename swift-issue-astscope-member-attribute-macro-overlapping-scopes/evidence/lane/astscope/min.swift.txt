func take(_ n: Int) -> [UInt8]? {
    var b = [UInt8](repeating: 0, count: n), i = 0
    while i < n {
        let r = b.withUnsafeMutableBytes { get($0.baseAddress! + i, n - i) }
        if r <= 0 { return nil }
        i += r
    }
    return b
}
func send(_ s: String) {
    var n = UInt64(s.utf8.count).littleEndian
    withUnsafeBytes(of: &n) { put($0.baseAddress!, 8) }
    Array(s.utf8).withUnsafeBytes { put($0.baseAddress!, $0.count) }
}
