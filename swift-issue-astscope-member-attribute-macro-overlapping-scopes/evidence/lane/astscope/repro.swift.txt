func readExactly(_ n: Int) -> [UInt8]? {
    var buffer = [UInt8](repeating: 0, count: n)
    var got = 0
    while got < n {
        let r = buffer.withUnsafeMutableBytes { readBytes($0.baseAddress! + got, n - got) }
        if r <= 0 { return nil }
        got += r
    }
    return buffer
}

func send(_ s: String) {
    let bytes = Array(s.utf8)
    var count = UInt64(bytes.count).littleEndian
    withUnsafeBytes(of: &count) { writeBytes($0.baseAddress!, 8) }
    bytes.withUnsafeBytes { writeBytes($0.baseAddress!, bytes.count) }
}
