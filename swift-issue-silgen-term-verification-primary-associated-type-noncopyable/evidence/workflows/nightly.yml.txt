name: nightly
on: workflow_dispatch
permissions:
  contents: read
jobs:
  linux:
    runs-on: ubuntu-latest
    container: swiftlang/swift:nightly-main-jammy
    steps:
      - uses: actions/checkout@v4
      - run: ./run.sh
  windows-nightly:
    runs-on: windows-latest
    steps:
      - uses: actions/checkout@v4
      - uses: SwiftyLab/setup-swift@38f54a76b70d989321de9dc7c840618c08cf56e9
        with:
          swift-version: https://download.swift.org/development/windows10/swift-DEVELOPMENT-SNAPSHOT-2026-10-17-a/swift-DEVELOPMENT-SNAPSHOT-2026-10-17-a-windows10.exe
      - run: swift --version
      - name: Variants
        shell: pwsh
        run: |
          foreach ($f in (Get-ChildItem v/*.swift, v2/*.swift, v3/*.swift)) {
            swiftc -parse-as-library -module-name M -package-name p -swift-version 6 -Onone -enable-experimental-feature Lifetimes -c $f.FullName -o "$($f.BaseName).obj" 2>&1 | Out-File "$($f.BaseName).log"
            $code = $LASTEXITCODE
            $first = (Select-String -Path "$($f.BaseName).log" -Pattern "Assertion failed|error:" | Select-Object -First 1).Line
            Get-Content "$($f.BaseName).log" | Select-String -Pattern "Assertion failed|Abort:|Initial term|hasErrorResult" | Select-Object -First 4 | ForEach-Object { Write-Output "  LOG $_" }
            Write-Output "RESULT $($f.Name) exit=$code $first"
          }
          exit 0
