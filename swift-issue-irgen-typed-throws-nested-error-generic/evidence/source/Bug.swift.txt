/// IRGen Crash: Closure with typed throws using nested Error in generic type
///
/// The compiler crashes (signal 6) during IR generation when a closure `{ $0 }`
/// is assigned to a stored property of type `(T) throws(Error) -> T` where `Error`
/// is a nested type of the enclosing generic struct, and T is concretized via
/// a constrained extension.
///
/// Assertion: `(hasErrorResult()), function getMutableErrorResult, file Types.h`
///
/// Conditions required (ALL must be present):
/// 1. Generic struct with a nested `Error` type
/// 2. Stored closure property typed as `throws(Error)` using the nested error
/// 3. Constrained extension (`where T == ConcreteType`) providing the closure
/// 4. swift.org open-source toolchain (+assertions build)
///
/// Note: Does NOT require @Sendable, cross-module, or ~Copyable.

// MARK: - Minimal Reproduction (8 lines)

public struct Box<T> {
    public var f: (T) throws(Error) -> T
    public enum Error: Swift.Error { case e }
}

public enum E { case a }

extension Box where T == E {
    public static var id: Self { .init(f: { $0 }) }  // CRASHES
}
