name: final
on:
  push:
    branches: [lane/astscope-member-attribute]
    paths: [final/**, .github/workflows/final.yml]
  workflow_dispatch:
permissions:
  contents: read
jobs:
  windows:
    runs-on: windows-latest
    steps:
      - uses: actions/checkout@3d3c42e5aac5ba805825da76410c181273ba90b1
      - uses: SwiftyLab/setup-swift@38f54a76b70d989321de9dc7c840618c08cf56e9
        with:
          swift-version: "6.4.1"
          cache-snapshot: false
      - shell: bash
        run: |
          final/run.sh windows-6.4.1
  macos:
    runs-on: macos-26
    steps:
      - uses: actions/checkout@3d3c42e5aac5ba805825da76410c181273ba90b1
      - name: Install swiftly and toolchains
        run: |
          curl -fsSL -o "$RUNNER_TEMP/swiftly.pkg" https://download.swift.org/swiftly/darwin/swiftly.pkg
          installer -pkg "$RUNNER_TEMP/swiftly.pkg" -target CurrentUserHomeDirectory
          "$HOME/.swiftly/bin/swiftly" init --assume-yes --skip-install --quiet-shell-followup
          for t in main-snapshot-2026-10-17 6.4.x-snapshot-2026-10-20 6.4.1; do "$HOME/.swiftly/bin/swiftly" install $t --assume-yes; done
      - name: Xcode
        run: |
          xcodebuild -version
          SWIFTC="xcrun swiftc" final/run.sh xcode
      - name: swiftly toolchains
        run: |
          for t in main-snapshot-2026-10-17 6.4.x-snapshot-2026-10-20 6.4.1; do
            TOOLCHAIN=$t SWIFTC="$PWD/astscope/swiftly-swiftc.sh" final/run.sh macos-$t
          done
  linux:
    runs-on: ubuntu-latest
    strategy:
      fail-fast: false
      matrix:
        image: ["swiftlang/swift:nightly-main-jammy", "swift:6.4.1-noble"]
    container: ${{ matrix.image }}
    steps:
      - uses: actions/checkout@3d3c42e5aac5ba805825da76410c181273ba90b1
      - run: |
          final/run.sh "linux-${{ matrix.image }}"
