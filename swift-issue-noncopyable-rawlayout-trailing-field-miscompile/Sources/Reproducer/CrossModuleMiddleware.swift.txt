// `Buffer` stores two generic `@_rawLayout` values from `Bug1Core`. The
// second field (`_b`) is the trailing field after the first raw-layout field
// (`_a`). Compiling this module at `-O` aborts in the LLVM verifier:
//
//   swiftc -swift-version 6 -O -wmo -parse-as-library \
//     -enable-experimental-feature RawLayout \
//     -I mods -module-name Bug1Middleware \
//     CrossModuleMiddleware.swift -c -o middleware.o
//
//   Instruction does not dominate all uses!
//     %14 = load i64, ptr %13, align 8, !invariant.load !17
//     %21 = mul i64 %14, 8
//
// Control: delete the `_b` field and its initialization. It compiles.
// `-Onone` also compiles. Concatenating both modules into one file (no
// import) still crashes, so the module split is not required on 6.4.2.

public import Bug1Core

public struct Buffer<Element: ~Copyable>: ~Copyable {
    var _a: Container<Element>.Inline<8>
    var _b: Container<Element>.Inline<4>

    public init() {
        self._a = Container<Element>.Inline<8>()
        self._b = Container<Element>.Inline<4>()
    }
}
