struct One<Value: Sendable>: Sendable {
    let _resume: @Sendable (sending () -> sending Value) async -> Void
    func resume(returning value: sending Value) async { await _resume { value } }
}
@main struct App { static func main() async {
    let one = One<String> { thunk in print(thunk()) }
    await one.resume(returning: "hello")
} }
